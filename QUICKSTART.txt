# ==================== BASIC USAGE ====================

# Import modules
from src.rdf_graph import RDFGraph
from src.neural_lp import NeuralLPHybrid

# 1. Create RDF Graph
graph = RDFGraph()

# 2. Add triples (Subject, Predicate, Object)
graph.add_triple('customer_1', 'buys', 'product_1')
graph.add_triple('customer_1', 'buys', 'product_2')
graph.add_triple('product_1', 'has_category', 'electronics')

# 3. Create Neural-LP model
model = NeuralLPHybrid(
    graph,
    embedding_dim=64,      # Dimension embedding
    alpha=0.5,            # Balance neural vs rules (0.5 = equal)
    rule_weight=0.4       # Weight for rules
)

# 4. Train model
model.train(
    num_epochs=10,        # Training iterations
    learning_rate=0.01,   # Learning rate
    mine_rules=True       # Extract rules
)

# 5. Make predictions
prediction = model.predict_link('customer_1', 'buys', 'product_3')
print(f"Confidence: {prediction.confidence:.2%}")
print(model.explain_prediction(prediction))

# 6. Rank candidates
candidates = ['product_3', 'product_4', 'product_5']
ranked = model.rank_candidates('customer_1', 'buys', candidates, top_k=3)

for i, pred in enumerate(ranked, 1):
    print(f"{i}. {pred.target} (confidence: {pred.confidence:.2%})")

# ==================== CONFIGURATION ====================

# Default Hyperparameters:

# embedding_dim: 64
#   - Dimension dari entity dan relation embeddings
#   - Smaller = faster training, less expressive
#   - Larger = slower training, more expressive
#   - Try: 32, 64, 128

# alpha: 0.5
#   - Balance antara neural dan rule-based scores
#   - 0.0 = pure rule-based
#   - 0.5 = equal weight
#   - 1.0 = pure neural
#   - Try: 0.3, 0.5, 0.7

# learning_rate: 0.01
#   - Step size untuk gradient descent
#   - Smaller = more stable, slower convergence
#   - Larger = faster but unstable
#   - Try: 0.001, 0.01, 0.1

# num_epochs: 10-20
#   - Training iterations
#   - More = better convergence, risk overfitting
#   - Try: 5, 10, 20, 50

# ==================== PERFORMANCE TIPS ====================

# 1. Use appropriate embedding_dim
#    - Small dataset (<100 entities): 32-64
#    - Medium dataset (100-1000): 64-128
#    - Large dataset (>1000): 128-256

# 2. Tune learning_rate
#    - Start with 0.01
#    - If loss oscillates, decrease to 0.001
#    - If convergence is slow, increase to 0.1

# 3. Adjust alpha based on your data
#    - More structured data (KG): use higher alpha (0.7)
#    - More transaction data: use lower alpha (0.3)
#    - Balance: use 0.5
