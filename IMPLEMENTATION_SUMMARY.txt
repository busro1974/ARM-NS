   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   - EntityEmbedding: Vector representation untuk entities

3. RULE MINING ENGINE (rule_mining.py)
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   - Rule: Representasi rule dalam bentuk body → head
   - SemanticMeasure: Traditional dan semantic measures
   - RuleMiningEngine: Main mining engine
   
   Key Methods:
   • mine_rules() → List[Rule]
   • mine_chain_rules(p1, p2, p3)
   • mine_composition_rules(p1, p2)
   • rank_rules(rules) → Ranked rules by score
   • get_top_rules(k)
   • get_rules_by_predicate(predicate)
   
   Features:
   ✓ Chain rule mining: p1(x,y) ∧ p2(y,z) -> p3(x,z)
   ✓ Composition rule mining: p1(x,y) -> p2(x,y)
   ✓ Semantic measures untuk rule quality
   ✓ Thresholding dengan min_support dan min_confidence

4. SEMANTIC INTERESTINGNESS MEASURE (semantic_interestingness.py)
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   - SemanticInterestingnessMeasure: 10+ quality metrics
   - PathAnalyzer: Path-based semantic analysis
   
   Metrics Tersedia:
   ✓ Support: P(body)
   ✓ Confidence: P(head|body)
   ✓ Lift: conf / support(head)
   ✓ Conviction: (1-support) / (1-confidence)
   ✓ Coherence: semantic relatedness via paths
   ✓ Informativeness: rule information content
   ✓ Specificity: body constraint level
   ✓ Relevance: domain alignment
   ✓ Consistency: exception handling
   ✓ Novelty: uniqueness compared to existing rules
   ✓ Semantic Cohesion: entity semantic relatedness
   
   Key Methods:
   • support(body_count, total_count)
   • confidence(body_head_count, body_count)
   • lift(confidence, head_support)
   • coherence(rule, path_metrics)
   • informativeness(confidence, support)
   • combined_interestingness(rule, metrics, weights)
   • get_path_metrics(source, target)

5. NEURAL-LP HYBRID SYSTEM (neural_lp.py)
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   - Prediction: Link prediction dengan evidence
   - NeuralLPHybrid: Main hybrid system
   
   Key Methods:
   • train(num_epochs, learning_rate, mine_rules)
   • predict_link(subject, relation, target) → Prediction
   • rank_candidates(subject, relation, candidates, top_k)
   • explain_prediction(prediction) → str
   • get_statistics() → Dict
   
   Hybrid Scoring Formula:
   combined_score = alpha * neural_score + (1-alpha) * rule_score
   confidence = sigmoid(combined_score)
   
   Features:
   ✓ Flexible weighting (alpha parameter)
   ✓ Configurable rule weight
   ✓ Prediction explanations dengan evidence
   ✓ Semantic rule evaluation
   ✓ Candidate ranking

   - EvaluationResult: Single domain evaluation result

USAGE EXAMPLES

BASIC USAGE
───────────────────────────────────────────────────────────────
from rdf_graph import RDFGraph
from neural_lp import NeuralLPHybrid

# Create graph
graph = RDFGraph()
graph.add_triple('customer_1', 'buys', 'product_1')
graph.add_triple('product_1', 'has_category', 'electronics')

# Create and train model
model = NeuralLPHybrid(graph, embedding_dim=64, alpha=0.5)
model.train(num_epochs=20, learning_rate=0.01, mine_rules=True)

# Make predictions
prediction = model.predict_link('customer_1', 'buys', 'product_2')
print(model.explain_prediction(prediction))

RANKING CANDIDATES
───────────────────────────────────────────────────────────────
candidates = ['product_2', 'product_3', 'product_4']
ranked = model.rank_candidates('customer_1', 'buys', candidates, top_k=3)

for i, pred in enumerate(ranked, 1):
    print(f"{i}. {pred.target} (confidence: {pred.confidence:.2%})")
